        // تحويل الرقم الإنجليزي إلى رقم عربي
        function convertToArabicNumber(char) {
          const arabicNumbers = ['٠','١','٢','٣','٤','٥','٦','٧','٨','٩'];
          return arabicNumbers[char] || char;
        }

        function insertAtCursor(input, char) {
          const start = input.selectionStart;
          const end = input.selectionEnd;
          const text = input.value;
          input.value = text.slice(0, start) + char + text.slice(end);
          input.selectionStart = input.selectionEnd = start + char.length;
        }

          function handleInput(event) {
          const charCode = event.which || event.keyCode;
          const char = String.fromCharCode(charCode);

          const isArabic = /^[\u0600-\u06FF]$/.test(char);
          const isEnglish = /^[a-zA-Z]$/.test(char);
          const isDigit = /^[0-9]$/.test(char);
          const isSpace = char === ' ';
          const isBackslash = char === '\\';
          const isForwardSlash = char === '/';

          // السماح بالمفاتيح الخاصة
          if (
            charCode === 8 ||  // Backspace
            charCode === 9 ||  // Tab
            charCode === 46 || // Delete
            (charCode >= 37 && charCode <= 40) // الأسهم
          ) {
            return;
          }

          // لو رقم → نحوله لرقم عربي
          if (isDigit) {
            event.preventDefault();
            insertAtCursor(event.target, convertToArabicNumber(char));
            return;
          }

          // لو حرف عربي أو إنجليزي أو مسافة أو باك سلاش أو سلاش → مسموح
          if (isArabic || isEnglish || isSpace || isBackslash || isForwardSlash) {
            return;
          }

          // غير كده → امنع الإدخال
          event.preventDefault();
        }

        function handlePaste(event) {
          const pasted = (event.clipboardData || window.clipboardData).getData('text');

          // استبدال الأرقام الإنجليزية بالأرقام العربية
          const arabicNumbers = ['٠','١','٢','٣','٤','٥','٦','٧','٨','٩'];
          const converted = pasted.replace(/[0-9]/g, d => arabicNumbers[d]);

          // تحقق أن المحتوى لا يحتوي رموز خاصة
          if (!/^[\u0600-\u06FFa-zA-Z ٠-٩]+$/.test(converted)) {
            event.preventDefault();
            return;
          }

          event.preventDefault();
          insertAtCursor(event.target, converted);
        }
    </script>
}
